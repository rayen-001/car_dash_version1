Implementation Plan: Booking Tags & Notes Filtering System
Overview
Add a structured tag system on top of the existing owner_remarks (TEXT) and damage_notes (TEXT) columns in the bookings table. Tags are short prefixed tokens (e.g. [VIP], [DAMAGE]) that are stored inline in the text fields, parsed in the frontend, and surfaced as interactive filter chips in the Bookings page and Vehicle History page. No new database tables are required.

Current State (What Already Exists)
Layer	What exists
DB	bookings.owner_remarks TEXT NULL — free-form text used for notes
DB	bookings.damage_notes TEXT NULL — used for return damage log with color prefix [GREEN]/[GRAY]/[RED]
DB	bookings.client_behavior_status TEXT NULL — stores raw behavior label
Types	Booking.damage_notes?: string, Booking.owner_remarks (not in global type yet)

Tag Storage Format
Tags are embedded at the start of owner_remarks as bracket-prefixed tokens:


owner_remarks = "[VIP][AGENCY] Long-term client from Tunisie Voyages. Paid cash."
owner_remarks = "[DAMAGE][LATE_RETURN] Door scratch left side. Returned 3 hours late."
owner_remarks = "[INSURANCE_CLAIM] File #4521 opened with STAR insurance."
NOTE

This zero-migration approach works because owner_remarks is TEXT with no constraints. The [TAG] format is distinct from the existing [GREEN]/[GRAY]/[RED] system used in damage_notes, so there is zero collision risk between both systems.

/** Extract all tag keys present in a remarks string */
export function extractTags(remarks: string | null | undefined): string[] {
  if (!remarks) return []
  return BOOKING_TAGS
    .filter(t => remarks.includes(t.key))
    .map(t => t.key)
}
/** Get the full BookingTag metadata for a key */
export function getTagMeta(key: string): BookingTag | undefined {
  return BOOKING_TAGS.find(t => t.key === key)
}
/** Toggle a tag inside a remarks string. Adds if absent, removes if present. */
export function toggleTag(remarks: string, tagKey: string): string {
  if (remarks.includes(tagKey)) {
    return remarks.replace(tagKey, '').replace(/\s{2,}/g, ' ').trim()
  }
  // Prepend tag, keep free text after
  return `${tagKey} ${remarks}`.trim()
}
/** Strip all known tags from a string, returning only the free text */
export function stripTags(remarks: string): string {
  let result = remarks
  for (const t of BOOKING_TAGS) {
    result = result.replace(t.key, '')
  }
  return result.replace(/\s{2,}/g, ' ').trim()
}

// src/components/TagPicker.tsx
'use client'
import { BOOKING_TAGS, extractTags, toggleTag } from '@/lib/bookingTags'
interface TagPickerProps {
  value: string           // current owner_remarks value
  onChange: (v: string) => void
}
export default function TagPicker({ value, onChange }: TagPickerProps) {
  const activeTags = extractTags(value)
  return (
    <div>
      <label style={{ fontSize: '0.8rem', color: 'var(--text-muted)', marginBottom: '0.5rem', display: 'block' }}>
        🏷️ Tags
      </label>
      <div style={{ display: 'flex', flexWrap: 'wrap', gap: '0.4rem' }}>
        {BOOKING_TAGS.map(tag => {
          const isActive = activeTags.includes(tag.key)
          return (
            <button
              key={tag.key}
              type="button"
              onClick={() => onChange(toggleTag(value, tag.key))}
              style={{
                padding: '0.25rem 0.65rem',
                borderRadius: '20px',
                fontSize: '0.78rem',
                fontWeight: 600,
                border: `1px solid ${isActive ? tag.border : 'rgba(255,255,255,0.1)'}`,
                background: isActive ? tag.bg : 'rgba(255,255,255,0.03)',
                color: isActive ? tag.color : 'rgba(255,255,255,0.45)',
                cursor: 'pointer',
                transition: 'all 0.15s ease',
              }}
            >
              {tag.emoji} {tag.label}
            </button>
          )
        })}
      </div>
    </div>
  )
}
Step 4 — Inline Tag Chips Component (Reusable)
[NEW FILE] src/components/TagChips.tsx

Used in table rows and history cards to render compact colored tags.

tsx

// src/components/TagChips.tsx
import { extractTags, getTagMeta } from '@/lib/bookingTags'
interface TagChipsProps {
  remarks: string | null | undefined
  compact?: boolean
}
export default function TagChips({ remarks, compact }: TagChipsProps) {
  const tags = extractTags(remarks)
  if (tags.length === 0) return null
  return (
    <div style={{ display: 'flex', flexWrap: 'wrap', gap: '0.25rem', marginTop: '0.3rem' }}>
      {tags.map(key => {
        const meta = getTagMeta(key)!
        return (
          <span
            key={key}
            title={meta.label}
            style={{
              padding: compact ? '0.1rem 0.4rem' : '0.2rem 0.5rem',
              borderRadius: '12px',
              fontSize: compact ? '0.68rem' : '0.73rem',
              fontWeight: 600,
              background: meta.bg,
              border: `1px solid ${meta.border}`,
              color: meta.color,
              whiteSpace: 'nowrap',
            }}
          >
            {meta.emoji} {compact ? '' : meta.label}
          </span>
        )
      })}
    </div>
  )
}
