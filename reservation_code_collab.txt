import pandas as pd
import numpy as np
import re
from datetime import timedelta

# ==========================================
# 📥 1. CONFIGURATION
# ==========================================
excel_file = "eline 2024-2025.xlsx"  # Esm el file mta3ek
sheet_name = "tous"                  # Sheet el fih el data

# Load el raw data
df = pd.read_excel(excel_file, sheet_name=sheet_name)

# Strip trailing spaces mel column names
df.columns = [str(col).strip() for col in df.columns]

print(f"📊 Fichier fih: {len(df)} lignes en total.")

# ==========================================
# ⚙️ 2. SAFE CLEANING FUNCTIONS
# ==========================================

# Standardize License Plate (Matricule)
def clean_plate(p):
    if pd.isna(p): 
        return ""
    p = str(p).upper().strip().replace(" ", "")
    p = p.replace("TY", "TU").replace("YU", "TU")
    return p

# Standardize Time to HH:MM format
def clean_time(t):
    if pd.isna(t): 
        return "10:00"
    t = str(t).lower().strip()
    if ":" in t:
        parts = t.split(":")
        try:
            return f"{int(parts[0]):02d}:{int(parts[1]):02d}"
        except:
            return "10:00"
    match = re.search(r"(\d+)\s*h\s*(\d*)", t)
    if match:
        h = int(match.group(1))
        m = int(match.group(2)) if match.group(2) else 0
        return f"{h:02d}:{m:02d}"
    return "10:00"

# Standardize and recalculate return date if 'nbr de jour' has additions
def calculate_dates(row):
    start_date = row.get('Date de départ')
    end_date = row.get('Date de retour')
    days_text = str(row.get('nbr de jour', '')).strip().lower()
    
    try:
        start_dt = pd.to_datetime(start_date)
    except:
        return None, None
        
    try:
        end_dt = pd.to_datetime(end_date)
    except:
        end_dt = start_dt + timedelta(days=1)
        
    if '+' in days_text:
        parts = days_text.split('+')
        total_days = 0
        for p in parts:
            p_digits = ''.join(filter(str.isdigit, p))
            if p_digits:
                total_days += int(p_digits)
        if total_days > 0:
            end_dt = start_dt + timedelta(days=total_days)
            
    return start_dt.strftime('%Y-%m-%d'), end_dt.strftime('%Y-%m-%d')

# Safely convert Acompte / Reste to numeric
def clean_numeric(val):
    if pd.isna(val): 
        return 0.0
    val_str = str(val).strip().replace(" ", "").replace(",", ".")
    match = re.search(r"[\d\.]+", val_str)
    if match:
        try:
            return float(match.group(0))
        except:
            return 0.0
    return 0.0

# ==========================================
# 🚀 3. DATA PROCESSING & SECURITY CHECK
# ==========================================

# Check if we actually have rows starting from index 4668 (row 4670 in Excel)
if len(df) <= 4668:
    print("⚠️  WARNING: El file Excel mta3ek fih 4668 lignes kahaw. Mafammech jdod!")
    print("👉 Béch nwarrik el structure, chna5thou a5er 10 lignes m9aydin fil file.")
    df_new = df.tail(10).copy()  # Tester les 10 dernières lignes pour voir le résultat
else:
    print(f"✅ Lqit {len(df) - 4668} nouvelles lignes.")
    df_new = df.iloc[4668:].copy()

# Recalculate Dates
cleaned_dates = []
for idx, row in df_new.iterrows():
    s_d, e_d = calculate_dates(row)
    cleaned_dates.append((s_d, e_d))

df_new['clean_start_date'] = [d[0] for d in cleaned_dates]
df_new['clean_end_date'] = [d[1] for d in cleaned_dates]

# Clean other columns safely
df_new['clean_plate'] = df_new['Immatriculation'].apply(clean_plate)
df_new['clean_pickup_time'] = df_new['Heure de départ'].apply(clean_time)
df_new['clean_return_time'] = df_new['Heure de retour'].apply(clean_time)
df_new['clean_acompte'] = df_new['ACOMPTE'].apply(clean_numeric)
df_new['clean_reste'] = df_new['RESTE'].apply(clean_numeric)
df_new['clean_total'] = df_new['clean_acompte'] + df_new['clean_reste']

# Map Remarque directly to damage_notes with [GRAY] prefix for grey box 💬
df_new['damage_notes'] = df_new['Remarque'].fillna("").apply(
    lambda x: f"[GRAY] {str(x).strip()}" if str(x).strip() else ""
)

df_new['clean_contract'] = df_new['N Contrat'].fillna("").apply(lambda x: str(x).split('.')[0].strip())

# Build the final clean DataFrame
final_cols = {
    'clean_contract': 'contract_number',
    'clean_plate': 'license_plate',
    'Num Permis': 'client_license_number',
    'clean_start_date': 'start_date',
    'clean_end_date': 'end_date',
    'clean_pickup_time': 'pickup_time',
    'clean_return_time': 'return_time',
    'clean_total': 'total_amount',
    'clean_acompte': 'acompte_paid',
    'damage_notes': 'damage_notes'  # <-- Mapped to damage_notes for grey comment box
}

df_final = df_new[list(final_cols.keys())].rename(columns=final_cols)

# ==========================================
# 💾 4. EXPORT TO CSV
# ==========================================
output_csv = "reservations_cleaned_to_import.csv"
df_final.to_csv(output_csv, index=False, sep=";")

print("="*50)
print(f"🎉 Done! Clean CSV created: {output_csv}")
print(f"📊 Total bookings processed: {len(df_final)}")
print("="*50)
